  test.only('child windows handl',async({browser})=>
  {
const context= await browser.newContext();
//creates a new isolated browser session/new browser window in playwright
const page= await context.newPage();
//in that new browser window/session we have created a new page on that new page we will automate further
const username= page.locator('#username');
await page.goto('https://rahulshettyacademy.com/loginpagePractise/');
const documentlink=page.locator("[href*='documents-request']");
const [newPage]= await Promise.all([//both are succesfully fulfilled and then jumps to the next,this is an Array index 
//needs to be parallelly executed 
context.waitForEvent('page'),//listen for any new page pending,rejecting,fulfilled
//another context for another page 
//waitforevent to monitor whether any page is opened in the background 
//the page2 variable will have the knowledge 
//we want the both to go parallell no await and async here 
documentlink.click(),
//expectations of this array is to return fulfilled promise and return array
])//we will come out of this array only when this promise box is fullfiled 
//the page2 variable will have the knowledge 
//these two steps will be iterated until the array index is fullfilled 
//and the fullfiled promise is a new page 
//NOTE: very important why is it asyncronous? why multiple steps
//hence promise is used as an array and the result is put into an object here which is [newpage]
//and then we move to the next steps
//everything happens inside the array once it is fullfilled 
//comes out of the array with the result and moves to the next steps
